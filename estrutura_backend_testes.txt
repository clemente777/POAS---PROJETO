from datetime import UTC, datetime
from typing import ClassVar

from fastapi import HTTPException
from sqlalchemy.exc import SQLAlchemyError

from backend.models.agendamento_model import Agendamentos
from backend.models.animal_model import Animais
from backend.models.usuario_model import Usuarios
from backend.repositories.agendamento_repository import AgendamentoRepository
from backend.schemas.agendamento_schema import AgendamentoCreate, AgendamentoUpdate


class AgendamentoServiceImpl:
    """
    Regras de negÃ³cio dos agendamentos.

    Regras:

    - Cliente agenda apenas seus animais.
    - VeterinÃ¡rio e Administrador possuem acesso total.
    - Animal precisa existir.
    - Data nÃ£o pode estar no passado.
    - NÃ£o permite conflito de horÃ¡rio.
    - Status possui valores controlados.
    - HistÃ³rico nÃ£o Ã© apagado.
    """

    STATUS_VALIDOS: ClassVar[list[str]] = [
        "Pendente",
        "Confirmado",
        "Cancelado",
        "Finalizado",
    ]

    def __init__(self, repository: AgendamentoRepository, usuario_logado: Usuarios):
        self.repository = repository
        self.usuario_logado = usuario_logado

    # ==========================================================
    # PERFIL
    # ==========================================================

    def obter_perfil(self):

        perfil = self.usuario_logado.perfil

        if hasattr(perfil, "nome"):
            return perfil.nome

        return perfil

    # ==========================================================
    # VERIFICAR ACESSO AO ANIMAL
    # ==========================================================

    def verificar_acesso_animal(self, animal: Animais):

        perfil = self.obter_perfil()

        # ------------------------------------------------------
        # ADMINISTRADOR E VETERINÃRIO
        # ------------------------------------------------------

        if perfil in ["Administrador", "VeterinÃ¡rio"]:
            return

        # ------------------------------------------------------
        # CLIENTE
        # ------------------------------------------------------

        if perfil == "Cliente":
            cliente = self.repository.buscar_cliente_por_usuario_id(
                self.usuario_logado.id
            )

            if not cliente:
                raise HTTPException(
                    status_code=403, detail=("UsuÃ¡rio nÃ£o possui cliente vinculado.")
                )

            if animal.cliente_id != cliente.id:
                raise HTTPException(
                    status_code=403, detail=("VocÃª nÃ£o possui acesso a este animal.")
                )

            return

        # ------------------------------------------------------
        # PERFIL INVÃLIDO
        # ------------------------------------------------------

        raise HTTPException(status_code=403, detail="Perfil sem permissÃ£o.")

    # ==========================================================
    # BUSCAR ANIMAL
    # ==========================================================

    def buscar_animal(self, animal_id: int):

        animal = self.repository.buscar_animal(animal_id)

        if not animal:
            raise HTTPException(status_code=404, detail="Animal nÃ£o encontrado.")

        self.verificar_acesso_animal(animal)

        return animal

    # ==========================================================
    # BUSCAR AGENDAMENTO POR ID
    # ==========================================================

    def buscar_por_id(self, id: int):

        agendamento = self.repository.buscar_por_id(id)

        if not agendamento:
            raise HTTPException(status_code=404, detail="Agendamento nÃ£o encontrado.")

        self.verificar_acesso_animal(agendamento.animal)

        return agendamento

    # ==========================================================
    # VALIDAÃ‡ÃƒO DE DATA
    # ==========================================================

    def validar_data(self, data: datetime):

        agora = datetime.now(UTC)

        if data.tzinfo is None:
            data = data.replace(tzinfo=UTC)

        if data < agora:
            raise HTTPException(
                status_code=400, detail=("NÃ£o Ã© permitido agendamento em data passada.")
            )

    # ==========================================================
    # VALIDAÃ‡ÃƒO DE STATUS
    # ==========================================================

    def validar_status(self, status: str):

        if status not in self.STATUS_VALIDOS:
            raise HTTPException(status_code=400, detail="Status invÃ¡lido.")

    # ==========================================================
    # VALIDAÃ‡ÃƒO DE DESCRIÃ‡ÃƒO
    # ==========================================================

    def validar_descricao(self, descricao: str):

        if not descricao:
            raise HTTPException(status_code=400, detail="DescriÃ§Ã£o obrigatÃ³ria.")

        descricao = descricao.strip()

        if not descricao:
            raise HTTPException(status_code=400, detail="DescriÃ§Ã£o invÃ¡lida.")

        if len(descricao) < 5:
            raise HTTPException(
                status_code=400,
                detail=("DescriÃ§Ã£o deve possuir no mÃ­nimo 5 caracteres."),
            )

    # ==========================================================
    # VALIDAÃ‡ÃƒO DE CONFLITO
    # ==========================================================

    def validar_conflito_horario(
        self, animal_id: int, data: datetime, agendamento_id: int | None = None
    ):

        existe = self.repository.buscar_conflito_horario(
            animal_id, data, agendamento_id
        )

        if existe:
            raise HTTPException(
                status_code=409,
                detail=("JÃ¡ existe agendamento para este animal neste horÃ¡rio."),
            )

    # ==========================================================
    # CRIAR AGENDAMENTO
    # ==========================================================

    def criar(self, agendamento: AgendamentoCreate):

        self.buscar_animal(agendamento.animal_id)

        self.validar_data(agendamento.data_agendamento)

        self.validar_descricao(agendamento.descricao)

        self.validar_conflito_horario(
            agendamento.animal_id, agendamento.data_agendamento
        )

        dados = agendamento.model_dump()

        # ------------------------------------------------------
        # VETERINÃRIO Ã‰ AUTOMATICAMENTE ASSOCIADO
        # ------------------------------------------------------

        if self.obter_perfil() == "VeterinÃ¡rio":
            dados["veterinario_id"] = self.usuario_logado.id

        novo = Agendamentos(**dados, status="Pendente")

        try:
            return self.repository.criar(novo)

        except SQLAlchemyError as erro:
            raise HTTPException(
                status_code=500,
                detail="Erro ao criar agendamento.",
            ) from erro

    # ==========================================================
    # VALIDAR VETERINÃRIO RESPONSÃVEL
    # ==========================================================

    def validar_veterinario_responsavel(self, veterinario_id: int | None):
        """
        Regras para definir quem atende o agendamento:

        - None remove o veterinÃ¡rio responsÃ¡vel.
        - VeterinÃ¡rio sÃ³ pode assumir o agendamento para si.
        - Administrador pode indicar qualquer veterinÃ¡rio.
        """

        if veterinario_id is None:
            return

        if (
            self.obter_perfil() == "VeterinÃ¡rio"
            and veterinario_id != self.usuario_logado.id
        ):
            raise HTTPException(
                status_code=403,
                detail=("VocÃª sÃ³ pode assumir o agendamento para vocÃª mesmo."),
            )

        veterinario = self.repository.buscar_veterinario_por_id(veterinario_id)

        if not veterinario:
            raise HTTPException(status_code=404, detail="VeterinÃ¡rio nÃ£o encontrado.")

        if veterinario.perfil != "VeterinÃ¡rio":
            raise HTTPException(
                status_code=400, detail=("UsuÃ¡rio informado nÃ£o Ã© veterinÃ¡rio.")
            )

    # ==========================================================
    # ATRIBUIR VETERINÃRIO
    # ==========================================================

    def atribuir_veterinario(self, id: int, veterinario_id: int):

        agendamento = self.buscar_por_id(id)

        veterinario = self.repository.buscar_veterinario_por_id(veterinario_id)

        if not veterinario:
            raise HTTPException(status_code=404, detail="VeterinÃ¡rio nÃ£o encontrado.")

        if veterinario.perfil != "VeterinÃ¡rio":
            raise HTTPException(
                status_code=400, detail=("UsuÃ¡rio informado nÃ£o Ã© veterinÃ¡rio.")
            )

        agendamento.veterinario_id = veterinario_id

        try:
            return self.repository.atualizar(agendamento)

        except SQLAlchemyError as erro:
            raise HTTPException(
                status_code=500,
                detail="Erro ao atribuir veterinÃ¡rio.",
            ) from erro

    # ==========================================================
    # LISTAR AGENDAMENTOS
    # ==========================================================

    def listar(
        self,
        skip: int = 0,
        limit: int = 10,
        animal_id: int | None = None,
        status: str | None = None,
        descricao: str | None = None,
        data: datetime | None = None,
        sort_by: str = "data_agendamento",
        order: str = "asc",
    ):

        perfil = self.obter_perfil()

        cliente_id = None

        # ------------------------------------------------------
        # CLIENTE SÃ“ VISUALIZA SEUS ANIMAIS
        # ------------------------------------------------------

        if perfil == "Cliente":
            cliente = self.repository.buscar_cliente_por_usuario_id(
                self.usuario_logado.id
            )

            if not cliente:
                raise HTTPException(status_code=403, detail="Cliente nÃ£o encontrado.")

            cliente_id = cliente.id

        # ------------------------------------------------------
        # STATUS
        # ------------------------------------------------------

        if status:
            self.validar_status(status)

        # ------------------------------------------------------
        # PAGINAÃ‡ÃƒO
        # ------------------------------------------------------

        skip = max(skip, 0)

        if limit <= 0:
            limit = 10

        limit = min(limit, 100)

        # ------------------------------------------------------
        # LISTAGEM
        # ------------------------------------------------------

        return self.repository.listar(
            skip=skip,
            limit=limit,
            animal_id=animal_id,
            status=status,
            descricao=(descricao.strip() if descricao else None),
            data=data,
            cliente_id=cliente_id,
            sort_by=sort_by,
            order=order,
        )

    # ==========================================================
    # ATUALIZAR AGENDAMENTO
    # ==========================================================

    def atualizar(self, id: int, dados: AgendamentoUpdate):

        agendamento = self.buscar_por_id(id)

        valores = dados.model_dump(exclude_unset=True)

        if not valores:
            raise HTTPException(
                status_code=400, detail=("Nenhum dado informado para atualizaÃ§Ã£o.")
            )

        # ------------------------------------------------------
        # DATA
        # ------------------------------------------------------

        if "data_agendamento" in valores:
            self.validar_data(valores["data_agendamento"])

            self.validar_conflito_horario(
                agendamento.animal_id, valores["data_agendamento"], agendamento.id
            )

        # ------------------------------------------------------
        # DESCRIÃ‡ÃƒO
        # ------------------------------------------------------

        if "descricao" in valores:
            self.validar_descricao(valores["descricao"])

        # ------------------------------------------------------
        # STATUS
        # ------------------------------------------------------

        if "status" in valores:
            self.validar_status(valores["status"])

        # ------------------------------------------------------
        # VETERINÃRIO RESPONSÃVEL
        # ------------------------------------------------------

        if "veterinario_id" in valores:
            self.validar_veterinario_responsavel(valores["veterinario_id"])

        # ------------------------------------------------------
        # CAMPOS PERMITIDOS
        # ------------------------------------------------------

        campos_permitidos = [
            "data_agendamento",
            "descricao",
            "status",
            "veterinario_id",
        ]

        for campo, valor in valores.items():
            if campo in campos_permitidos:
                setattr(agendamento, campo, valor)

        try:
            return self.repository.atualizar(agendamento)

        except SQLAlchemyError as erro:
            raise HTTPException(
                status_code=500,
                detail="Erro ao atualizar agendamento.",
            ) from erro

    # ==========================================================
    # CANCELAR AGENDAMENTO
    # ==========================================================

    def cancelar(self, id: int):

        agendamento = self.buscar_por_id(id)

        if agendamento.status == "Finalizado":
            raise HTTPException(
                status_code=409,
                detail=("NÃ£o Ã© possÃ­vel cancelar atendimento finalizado."),
            )

        if agendamento.status == "Cancelado":
            raise HTTPException(
                status_code=409, detail=("Agendamento jÃ¡ estÃ¡ cancelado.")
            )

        agendamento.status = "Cancelado"

        try:
            return self.repository.cancelar(agendamento)

        except SQLAlchemyError as erro:
            raise HTTPException(
                status_code=500,
                detail="Erro ao cancelar agendamento.",
            ) from erro

    # ==========================================================
    # DELETAR AGENDAMENTO
    #
    # BLOQUEADO
    # ==========================================================

    def deletar(self, id: int):

        self.repository.buscar_por_id(id)

        raise HTTPException(
            status_code=409,
            detail=("Agendamentos nÃ£o podem ser excluÃ­dos. Utilize o cancelamento."),
        )

from fastapi import HTTPException

from backend.models.animal_model import Animais
from backend.models.cliente_model import Clientes
from backend.models.usuario_model import Usuarios
from backend.repositories.animal_repository import AnimalRepository
from backend.schemas.animal_schema import AnimalCreate, AnimalUpdate


class AnimalServiceImpl:
    def __init__(self, repository: AnimalRepository, usuario_logado: Usuarios):

        self.repository = repository
        self.usuario_logado = usuario_logado

    # ==========================================================
    # AUXILIAR
    # ==========================================================

    def obter_perfil(self):

        return self.usuario_logado.perfil

    # ==========================================================
    # NORMALIZAÃ‡ÃƒO
    # ==========================================================

    def normalizar_texto(self, texto: str) -> str:

        if not texto:
            return ""

        return " ".join(texto.strip().split()).title()

    # ==========================================================
    # VALIDAÃ‡Ã•ES
    # ==========================================================

    def validar_nome(self, nome: str) -> bool:

        if not nome:
            return False

        return len(nome.strip()) >= 2

    def validar_especie(self, especie: str) -> bool:

        if not especie:
            return False

        return len(especie.strip()) >= 2

    def validar_raca(self, raca: str) -> bool:

        if not raca:
            return False

        return len(raca.strip()) >= 2

    def validar_idade(self, idade: int) -> bool:

        if idade is None:
            return False

        return 0 <= idade <= 100

    # ==========================================================
    # VALIDAÃ‡ÃƒO DOS CAMPOS
    # ==========================================================

    def validar_campos_obrigatorios(self, animal: AnimalCreate) -> None:

        if not self.validar_nome(animal.nome):
            raise HTTPException(status_code=400, detail="Nome invÃ¡lido.")

        if not self.validar_especie(animal.especie):
            raise HTTPException(status_code=400, detail="EspÃ©cie invÃ¡lida.")

        if not self.validar_raca(animal.raca):
            raise HTTPException(status_code=400, detail="RaÃ§a invÃ¡lida.")

        if not self.validar_idade(animal.idade):
            raise HTTPException(status_code=400, detail="Idade invÃ¡lida.")

    # ==========================================================
    # BUSCAS
    # ==========================================================

    def buscar_animal_por_id(self, animal_id: int) -> Animais:

        animal = self.repository.buscar_por_id(animal_id)

        if not animal:
            raise HTTPException(status_code=404, detail="Animal nÃ£o encontrado.")

        return animal

    def buscar_cliente_por_id(self, cliente_id: int) -> Clientes:

        cliente = self.repository.buscar_cliente_por_id(cliente_id)

        if not cliente:
            raise HTTPException(status_code=404, detail="Cliente nÃ£o encontrado.")

        return cliente

    # ==========================================================
    # CONTROLE DE PERMISSÃƒO
    # ==========================================================

    def verificar_acesso_animal(self, animal: Animais) -> None:

        perfil = self.obter_perfil()

        # Administrador e VeterinÃ¡rio
        # possuem acesso total.

        if perfil in ["Administrador", "VeterinÃ¡rio"]:
            return

        # Cliente somente acessa seus prÃ³prios animais.

        if perfil == "Cliente":
            cliente = self.repository.buscar_cliente_por_usuario_id(
                self.usuario_logado.id
            )

            if not cliente:
                raise HTTPException(
                    status_code=403, detail="UsuÃ¡rio sem cliente vinculado."
                )

            if animal.cliente_id != cliente.id:
                raise HTTPException(
                    status_code=403,
                    detail=("VocÃª nÃ£o possui permissÃ£o para acessar este animal."),
                )

            return

        raise HTTPException(status_code=403, detail="Perfil sem permissÃ£o.")

    # ==========================================================
    # CRIAR
    # ==========================================================

    def criar(self, animal: AnimalCreate) -> Animais:

        self.validar_campos_obrigatorios(animal)

        cliente = self.buscar_cliente_por_id(animal.cliente_id)

        perfil = self.obter_perfil()

        # Cliente sÃ³ pode cadastrar animal para si mesmo.

        if perfil == "Cliente":
            cliente_usuario = self.repository.buscar_cliente_por_usuario_id(
                self.usuario_logado.id
            )

            if not cliente_usuario:
                raise HTTPException(
                    status_code=403, detail="UsuÃ¡rio sem cliente vinculado."
                )

            if cliente_usuario.id != cliente.id:
                raise HTTPException(
                    status_code=403,
                    detail=("VocÃª nÃ£o pode cadastrar animal para outro cliente."),
                )

        novo_animal = Animais(
            nome=self.normalizar_texto(animal.nome),
            especie=self.normalizar_texto(animal.especie),
            raca=self.normalizar_texto(animal.raca),
            idade=animal.idade,
            cliente_id=cliente.id,
        )

        return self.repository.criar(novo_animal)

    # ==========================================================
    # LISTAR
    # ==========================================================

    def listar(
        self,
        pagina: int = 1,
        limite: int = 10,
        nome: str | None = None,
        ordem: str = "asc",
    ) -> list[Animais]:

        perfil = self.obter_perfil()

        # Corrige paginaÃ§Ã£o invÃ¡lida.

        pagina = max(pagina, 1)

        if limite < 1:
            limite = 10

        cliente_id = None

        # Cliente sÃ³ pode visualizar seus animais.

        if perfil == "Cliente":
            cliente = self.repository.buscar_cliente_por_usuario_id(
                self.usuario_logado.id
            )

            if not cliente:
                raise HTTPException(
                    status_code=403, detail="UsuÃ¡rio sem cliente vinculado."
                )

            cliente_id = cliente.id

        elif perfil not in ["Administrador", "VeterinÃ¡rio"]:
            raise HTTPException(status_code=403, detail="Perfil sem permissÃ£o.")

        return self.repository.listar(
            pagina=pagina, limite=limite, nome=nome, ordem=ordem, cliente_id=cliente_id
        )

    # ==========================================================
    # BUSCAR POR ID
    # ==========================================================

    def buscar_por_id(self, animal_id: int) -> Animais:

        animal = self.buscar_animal_por_id(animal_id)

        self.verificar_acesso_animal(animal)

        return animal

    # ==========================================================
    # ATUALIZAR
    # ==========================================================

    def atualizar(self, animal_id: int, dados: AnimalUpdate) -> Animais:

        animal = self.buscar_animal_por_id(animal_id)

        self.verificar_acesso_animal(animal)

        dados_dict = dados.model_dump(exclude_unset=True)

        # ------------------------------------------------------
        # NOME
        # ------------------------------------------------------

        if "nome" in dados_dict:
            if not self.validar_nome(dados.nome):
                raise HTTPException(status_code=400, detail="Nome invÃ¡lido.")

            animal.nome = self.normalizar_texto(dados.nome)

        # ------------------------------------------------------
        # ESPÃ‰CIE
        # ------------------------------------------------------

        if "especie" in dados_dict:
            if not self.validar_especie(dados.especie):
                raise HTTPException(status_code=400, detail="EspÃ©cie invÃ¡lida.")

            animal.especie = self.normalizar_texto(dados.especie)

        # ------------------------------------------------------
        # RAÃ‡A
        # ------------------------------------------------------

        if "raca" in dados_dict:
            if not self.validar_raca(dados.raca):
                raise HTTPException(status_code=400, detail="RaÃ§a invÃ¡lida.")

            animal.raca = self.normalizar_texto(dados.raca)

        # ------------------------------------------------------
        # IDADE
        # ------------------------------------------------------

        if "idade" in dados_dict:
            if not self.validar_idade(dados.idade):
                raise HTTPException(status_code=400, detail="Idade invÃ¡lida.")

            animal.idade = dados.idade

        return self.repository.atualizar(animal)

    # ==========================================================
    # DELETAR
    # ==========================================================

    def deletar(self, animal_id: int):

        animal = self.buscar_animal_por_id(animal_id)

        self.verificar_acesso_animal(animal)

        possui_atendimento = self.repository.possui_atendimento(animal.id)

        if possui_atendimento:
            raise HTTPException(
                status_code=400,
                detail=("NÃ£o Ã© possÃ­vel excluir animal com histÃ³rico de atendimento."),
            )

        self.repository.deletar(animal)

        return {"message": "Animal removido com sucesso."}

            ) from erro

        except SQLAlchemyError as erro:
            raise HTTPException(

            ) from erro

class ItemCarrinhoServiceImpl:
    def __init__(self, repository: ItemCarrinhoRepository, usuario_logado: Usuarios):
        self.repository = repository
        self.usuario_logado = usuario_logado

    # ==================================================
    # VALIDAR PERFIL
    # ==================================================

    def validar_perfil_permitido(self):
        if self.usuario_logado.perfil not in ["Administrador", "Cliente"]:
            raise HTTPException(
                status_code=403,
                detail=(
                    "Apenas clientes e administradores "
                    "podem acessar os itens do carrinho."
                ),
            )

    # ==================================================
    # VALIDAR PROPRIETÃRIO
    # ==================================================

    def validar_proprietario(self, carrinho: Carrinhos):
        # Administrador pode acessar qualquer carrinho
        if self.usuario_logado.perfil == "Administrador":
            return

        # Cliente precisa ter um carrinho associado
        if not carrinho.cliente:
            raise HTTPException(
                status_code=404, detail="Cliente do carrinho nÃ£o encontrado."
            )

        # Cliente sÃ³ pode acessar o prÃ³prio carrinho
        if carrinho.cliente.usuario_id != self.usuario_logado.id:
            raise HTTPException(
                status_code=403,
                detail=("VocÃª nÃ£o possui permissÃ£o para acessar este carrinho."),
            )

    # ==================================================
    # BUSCAR ITEM POR ID
    # ==================================================

    def buscar_por_id(self, id: int):
        self.validar_perfil_permitido()

        item = self.repository.buscar_por_id(id)

        if not item:
            raise HTTPException(status_code=404, detail="Item nÃ£o encontrado.")

        self.validar_proprietario(item.carrinho)

        return item

    # ==================================================
    # BUSCAR CARRINHO
    # ==================================================

    def buscar_carrinho(self, carrinho_id: int):
        self.validar_perfil_permitido()

        carrinho = self.repository.buscar_carrinho(carrinho_id)

        if not carrinho:
            raise HTTPException(status_code=404, detail="Carrinho nÃ£o encontrado.")

        self.validar_proprietario(carrinho)

        return carrinho

    # ==================================================
    # BUSCAR PRODUTO
    # ==================================================

    def buscar_produto(self, produto_id: int):
        produto = self.repository.buscar_produto(produto_id)

        if not produto:
            raise HTTPException(status_code=404, detail="Produto nÃ£o encontrado.")

        return produto

    # ==================================================
    # CRIAR ITEM NO CARRINHO
    # ==================================================

    def criar(self, item: ItemCarrinhoCreate):
        """
        Adiciona um produto ao carrinho.

        Regras:
        - Apenas Cliente e Administrador podem adicionar.
        - O carrinho precisa existir.
        - O usuÃ¡rio precisa ter permissÃ£o sobre o carrinho.
        - O produto precisa existir.
        - A quantidade deve ser maior que zero.
        - A quantidade nÃ£o pode ultrapassar o estoque.
        - Se o produto jÃ¡ estiver no carrinho,
          a quantidade serÃ¡ somada.
        """

        self.validar_perfil_permitido()

        # --------------------------------------------------
        # VALIDAR QUANTIDADE
        # --------------------------------------------------

        if item.quantidade <= 0:
            raise HTTPException(
                status_code=400, detail="Quantidade deve ser maior que zero."
            )

        # --------------------------------------------------
        # BUSCAR CARRINHO E VALIDAR ACESSO
        # --------------------------------------------------

        self.buscar_carrinho(item.carrinho_id)

        # --------------------------------------------------
        # BUSCAR PRODUTO
        # --------------------------------------------------

        produto = self.buscar_produto(item.produto_id)

        # --------------------------------------------------
        # VALIDAR ESTOQUE
        # --------------------------------------------------

        if item.quantidade > produto.estoque:
            raise HTTPException(
                status_code=400, detail=("Quantidade maior que estoque disponÃ­vel.")
            )

        # --------------------------------------------------
        # VERIFICAR SE O PRODUTO JÃ ESTÃ NO CARRINHO
        # --------------------------------------------------

        item_existente = self.repository.buscar_item_existente(
            carrinho_id=item.carrinho_id, produto_id=item.produto_id
        )

        # --------------------------------------------------
        # PRODUTO JÃ EXISTE
        # --------------------------------------------------

        if item_existente:
            nova_quantidade = item_existente.quantidade + item.quantidade

            if nova_quantidade > produto.estoque:
                raise HTTPException(
                    status_code=400, detail=("Quantidade total ultrapassa estoque.")
                )

            item_existente.quantidade = nova_quantidade

            return self.repository.atualizar(item_existente)

        # --------------------------------------------------
        # CRIAR NOVO ITEM
        # --------------------------------------------------

        novo_item = ItensCarrinho(**item.model_dump())

        return self.repository.criar(novo_item)

    # ==================================================
    # LISTAR ITENS
    # ==================================================

    def listar(
        self,
        skip: int = 0,
        limit: int = 10,
        carrinho_id: int | None = None,
        produto_id: int | None = None,
        quantidade_min: int | None = None,
        quantidade_max: int | None = None,
        sort_by: str = "id",
        order: str = "asc",
    ):

            sort_by=sort_by,
            order=order,
        )

    # ==================================================
    # ATUALIZAR ITEM
    # ==================================================

    def atualizar(self, id: int, item: ItemCarrinhoUpdate):
        self.validar_perfil_permitido()

        item_db = self.buscar_por_id(id)

        dados = item.model_dump(exclude_unset=True)

        # --------------------------------------------------
        # NÃƒO HÃ NENHUM DADO PARA ATUALIZAR
        # --------------------------------------------------

        if not dados:
            raise HTTPException(
                status_code=400, detail="Nenhum dado foi informado para atualizaÃ§Ã£o."
            )

        # --------------------------------------------------
        # ATUALIZAR QUANTIDADE
        # --------------------------------------------------

        if "quantidade" in dados:
            quantidade = dados["quantidade"]

            if quantidade <= 0:

            raise

    def deletar(self, carrinho: Carrinhos):
        try:
            self.session.delete(carrinho)

            self.session.commit()

            return True

        except Exception:
            self.session.rollback()

            raise

    # ==================================================
    # FINALIZAR COMPRA
    # ==================================================

    def finalizar_compra(self, carrinho: Carrinhos):
        """
        Executa a operaÃ§Ã£o de finalizaÃ§Ã£o da compra.

        Responsabilidades do Repository:
        - Baixar o estoque dos produtos
        - Registrar a compra (pedido + itens) como histÃ³rico
        - Remover os itens do carrinho
        - Confirmar a transaÃ§Ã£o (tudo ou nada)
        - Desfazer a transaÃ§Ã£o em caso de erro

        As validaÃ§Ãµes de negÃ³cio permanecem no Service.
        """

        try:
            valor_total = 0.0
            quantidade_itens = 0

            itens_pedido = []

            # --------------------------------------------------
            # BAIXAR ESTOQUE, CALCULAR TOTAL E MONTAR HISTÃ“RICO
            # --------------------------------------------------

            for item in list(carrinho.itens):
                produto = item.produto

                # Baixa condicional: sÃ³ desconta se ainda houver
                # estoque suficiente. Evita estoque negativo se
                # duas compras acontecerem ao mesmo tempo.

                resultado = self.session.execute(
                    update(Produtos)
                    .where(
                        Produtos.id == produto.id, Produtos.estoque >= item.quantidade
                    )
                    .values(estoque=(Produtos.estoque - item.quantidade))
                )

                if resultado.rowcount == 0:
                    raise EstoqueInsuficienteError(produto.nome)

                valor_total += produto.preco * item.quantidade

                quantidade_itens += item.quantidade

                # Nome e preÃ§o sÃ£o copiados agora: o histÃ³rico
                # nÃ£o muda se o produto for editado depois.

                itens_pedido.append(
                    ItensPedido(
                        produto_id=produto.id,
                        nome_produto=produto.nome,
                        preco_unitario=produto.preco,
                        quantidade=item.quantidade,
                    )
                )

            valor_total = round(valor_total, 2)

            # --------------------------------------------------
            # REGISTRAR A COMPRA (PEDIDO)
            # --------------------------------------------------

            pedido = PedidoRepository(self.session).adicionar(
                Pedidos(
                    cliente_id=carrinho.cliente_id,
                    valor_total=valor_total,
                    itens=itens_pedido,
                )
            )

            pedido_id = pedido.id

            # --------------------------------------------------
            # REMOVER ITENS DO CARRINHO
            # --------------------------------------------------

            for item in list(carrinho.itens):
                self.session.delete(item)

            # --------------------------------------------------
            # CONFIRMAR TRANSAÃ‡ÃƒO
            # --------------------------------------------------

            self.session.commit()

            return {
                "mensagem": "Compra finalizada com sucesso.",
                "valor_total": valor_total,
                "quantidade_itens": quantidade_itens,
                "pedido_id": pedido_id,
            }

        except Exception:
            self.session.rollback()

            raise

    # ==================================================
    # ROLLBACK
    # ==================================================

    def rollback(self):
        """
        Desfaz a transaÃ§Ã£o atual.

        Usado pelo Service quando ocorre uma
        HTTPException durante uma operaÃ§Ã£o.
        """

        self.session.rollback()

    def listar(
        self,
        skip: int = 0,
        limit: int = 10,
        nome: str | None = None,
        cpf: str | None = None,
        email: str | None = None,
        usuario_id: int | None = None,
        sort_by: str = "nome",
        order: str = "asc",
    ):
        query = select(Clientes)

        if usuario_id is not None:
            query = query.where(Clientes.usuario_id == usuario_id)

        if nome:
            query = query.where(Clientes.nome.ilike(f"%{nome.strip()}%"))

        if cpf:
            query = query.where(Clientes.cpf == cpf)

        if email:
            query = query.where(Clientes.email.ilike(f"%{email.strip()}%"))

        campos = {
            "id": Clientes.id,
            "nome": Clientes.nome,
            "cpf": Clientes.cpf,
            "email": Clientes.email,
        }

        coluna = campos.get(sort_by, Clientes.nome)

        if order.lower() == "desc":
            query = query.order_by(desc(coluna))
        else:
            query = query.order_by(asc(coluna))

        return self.session.scalars(query.offset(skip).limit(limit)).all()

        )

        return self.session.scalar(select(Pedidos).where(Pedidos.id == pedido_id))

        self, skip: int = 0, limite: int = 10, cliente_id: int | None = None
    ) -> list[Pedidos]:

        # Filtro por cliente
        if cliente_id is not None:

    def adicionar(self, pedido: Pedidos) -> Pedidos:
        """
        Adiciona o pedido Ã  sessÃ£o e faz flush
        (para o id ficar disponÃ­vel), SEM commit.

        controla a transaÃ§Ã£o: baixa de estoque,
        registro do pedido e limpeza do carrinho
        sÃ£o confirmados juntos, ou nenhum deles.
        """

        self.session.add(pedido)
        self.session.flush()

# ==========================================================
# DEPENDÃŠNCIAS
# ==========================================================

SessionDep = Annotated[Session, Depends(get_session)]

LoginFormDep = Annotated[OAuth2PasswordRequestForm, Depends()]

# SERVICE
# ==========================================================


def get_usuario_service(session: SessionDep):
    usuario_repository = UsuarioRepository(session)
    cliente_repository = ClienteRepository(session)

    return UsuarioServiceImpl(
        repository=usuario_repository,
        cliente_repository=cliente_repository,
    )

    UsuarioServiceImpl,

# LOGIN

@router.post("/")
def login(
    form_data: LoginFormDep,
    service: UsuarioServiceDep,
):
    usuario = service.buscar_por_email(form_data.username)

    if not usuario:
        raise HTTPException(
            status_code=401,
            detail="UsuÃ¡rio nÃ£o encontrado.",
        )

    senha_valida = senha_context.verify(
        form_data.password,
        usuario.senha_hash,
    )

    if not senha_valida:
        raise HTTPException(
            status_code=401,
            detail="Senha invÃ¡lida.",
        )

    # ----------------------------------------------------------
    # VERIFICA SE CLIENTE POSSUI CADASTRO DE CLIENTE
    # ----------------------------------------------------------

    if usuario.perfil == "Cliente":
        cliente = service.buscar_cliente_por_usuario_id(usuario.id)

        if not cliente:
            raise HTTPException(
                status_code=404,
                detail="Este usuÃ¡rio nÃ£o possui cadastro de cliente.",
            )

    # ----------------------------------------------------------
    # GERA TOKEN
    # ----------------------------------------------------------

    token = create_access_token(
        {
            "sub": usuario.email,
            "perfil": usuario.perfil,
            "usuario_id": usuario.id,
        }
    )

    return {
        "access_token": token,
        "token_type": "bearer",
        "usuario": {
            "id": usuario.id,
            "nome": usuario.nome,
            "email": usuario.email,
            "perfil": usuario.perfil,
        },
    }

from datetime import UTC, datetime
from typing import Annotated

from fastapi import APIRouter, Depends, HTTPException
from fastapi.security import OAuth2PasswordBearer
from sqlalchemy.orm import Session

from backend.auth.token import decode_token
from backend.database.database import get_session
from backend.repositories.token_repository import TokenRepository
from backend.services.implementations.token_service_impl import TokenService

router = APIRouter(prefix="/logout", tags=["Logout"])

oauth2_scheme = OAuth2PasswordBearer(tokenUrl="/login/")

SessionDep = Annotated[Session, Depends(get_session)]
TokenDep = Annotated[str, Depends(oauth2_scheme)]


def get_token_service(session: SessionDep):
    repository = TokenRepository(session)

    return TokenService(repository=repository)


TokenServiceDep = Annotated[
    TokenService,
    Depends(get_token_service),
]


@router.post("/")
def logout(
    token: TokenDep,
    token_service: TokenServiceDep,
):
    # ==========================================
    # DECODIFICAR TOKEN
    # ==========================================

    try:
        payload = decode_token(token)

    except Exception as erro:
        raise HTTPException(
            status_code=401,
            detail="Token invÃ¡lido.",
        ) from erro

    # ==========================================
    # VALIDAR EXPIRAÃ‡ÃƒO
    # ==========================================

    if "exp" not in payload:
        raise HTTPException(status_code=401, detail="Token invÃ¡lido.")

    # ==========================================
    # CONVERTER EXPIRAÃ‡ÃƒO
    # ==========================================

    expira_em = datetime.fromtimestamp(payload["exp"], tz=UTC)

    # ==========================================
    # REVOGAR TOKEN
    # ==========================================

    token_service.revogar(token=token, expira_em=expira_em)

    # ==========================================
    # RETORNAR RESPOSTA
    # ==========================================

    return {"message": "Logout realizado com sucesso."}
